#!/bin/sh
if [[ $(< /var/local/refresh) == true ]]; then 
set -x
cd mnt
rm -rf *


set +x
resultAsJson=$(curl -u ${MAVEN_REPO_USER}:${MAVEN_REPO_PASS} -X POST  https://infyartifactory.ad.infosys.com/artifactory/api/search/aql -H "content-type: text/plain" -d 'items.find({ "repo": {"$eq":"icets-ai_npm"}, "name": {"$match" : "icip_app-3.0*.tgz"}})')
latestFile=$(echo $resultAsJson | jq -r '.results | sort_by(.updated) [length-1].name')
uiVersion=$latestFile
set -x
wget -q -N -P ./app/ui/ --user ${MAVEN_REPO_USER} --password ${MAVEN_REPO_PASS} https://infyartifactory.ad.infosys.com:443/artifactory/icets-ai_npm/icip_app/-/$latestFile 

tar -xf ./app/ui/icip_app-3.0*.tgz -C app/ui/
rm -rf ./app/icip_app-3.0-SNAPSHOT/ui/*
mv ./app/ui/package/* app/ui/

chmod -R 777 ./app
cd ..
fi
ls mnt

echo $uiVersion
